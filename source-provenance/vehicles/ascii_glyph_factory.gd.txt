class_name GlyphFactory
extends RefCounted
## Shared, genuinely extruded font geometry. Characters remain mesh objects at
## every camera angle; the top-down introduction simply conceals their depth.

static var _fonts: Dictionary = {}
static var _meshes: Dictionary = {}
static var _materials: Dictionary = {}

static func material(color: Color, emission: float = 0.0) -> StandardMaterial3D:
	var key: String = "%s/%.3f" % [color.to_html(true), emission]
	if _materials.has(key):
		return _materials[key] as StandardMaterial3D
	var result := StandardMaterial3D.new()
	result.albedo_color = color
	result.roughness = 0.72
	result.metallic = 0.16
	if color.a < 0.999:
		result.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
	if emission > 0.0:
		result.emission_enabled = true
		result.emission = Color(color.r, color.g, color.b, 1.0)
		result.emission_energy_multiplier = emission
	_materials[key] = result
	return result

static func glyph(text: String, color: Color, size: float = 1.0, depth: float = 0.2, font_style: String = "mono") -> MeshInstance3D:
	var result := MeshInstance3D.new()
	result.mesh = mesh(text, size, depth, font_style)
	result.material_override = material(color, 0.38)
	return result

static func font_for(style: String = "mono", text: String = "") -> Font:
	if not _fonts.has(style):
		var path: String = "res://assets/DejaVuSansMono-Bold.ttf"
		if style == "sans":
			path = "res://assets/DejaVuSans-Bold.ttf"
		elif style == "serif":
			path = "res://assets/DejaVuSerif-Bold.ttf"
		var loaded: Font = load(path) as Font
		_fonts[style] = loaded if loaded != null else ThemeDB.fallback_font
	var selected: Font = _fonts[style] as Font
	for index: int in text.length():
		if not selected.has_char(text.unicode_at(index)) and style != "sans":
			# Some extended letters and symbols exist only in bundled Sans.
			# Use its real outline instead of a system-dependent missing box.
			return font_for("sans", text)
	return selected

static func mesh(text: String, size: float = 1.0, depth: float = 0.2, font_style: String = "mono") -> TextMesh:
	var key: String = "%s/%.3f/%.3f/%s" % [text, size, depth, font_style]
	if _meshes.has(key):
		return _meshes[key] as TextMesh
	var result := TextMesh.new()
	result.font = font_for(font_style, text)
	result.font_size = 96
	result.pixel_size = size / 72.0
	result.depth = depth
	result.text = text
	result.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	result.vertical_alignment = VERTICAL_ALIGNMENT_CENTER
	result.curve_step = 0.08
	# Center the actual visible outline, including punctuation, rather than its
	# font ascender box. Depth is already centered by TextMesh itself.
	var bounds: AABB = result.get_aabb()
	result.offset = Vector2(-bounds.get_center().x, -bounds.get_center().y) / result.pixel_size
	_meshes[key] = result
	return result
